{{/* Renders a download button for a resume stored in the static/ folder. Usage: {{< resume-button >}} or {{< resume-button file="cv.pdf" label="Download CV" >}} If the file does not exist yet, nothing is rendered and a build warning is printed. */}} {{ $file := .Get "file" | default "resume.pdf" }} {{ $label := .Get "label" | default "Download Resume (PDF)" }} {{ if fileExists (printf "static/%s" $file) }} {{ partial "icon" "download" }} {{ $label }} {{ else }} {{ warnf "resume-button: static/%s not found, button hidden. See BLOG_MAINTENANCE.md." $file }} {{ end }}
